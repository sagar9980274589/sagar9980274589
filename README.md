> **Reflection isn't just a Java feature. It's how I evolve.**

---

```java
import java.lang.annotation.*;
import java.lang.reflect.Field;

@Retention(RetentionPolicy.RUNTIME)
@Target(ElementType.TYPE)
@interface Author {
    String value();
}

@Author("Sagar H D")
public class SagarHD {

    private String mindset = "Always Learning";
    private String focus = "Scalable Systems";
    private String strength = "Problem Solving";
    private String goal = "Continuous Growth";

    public static void main(String[] args) throws Exception {

        SagarHD me = new SagarHD();

        Author author = me.getClass().getAnnotation(Author.class);
        System.out.println("Author -> " + author.value());

        for (Field field : me.getClass().getDeclaredFields()) {
            field.setAccessible(true);
            System.out.println(
                field.getName() + " -> " + field.get(me)
            );
        }

        System.out.println(
            "\nReflection isn't just a Java feature."
            + "\nIt's how I evolve."
        );
    }
}
```
